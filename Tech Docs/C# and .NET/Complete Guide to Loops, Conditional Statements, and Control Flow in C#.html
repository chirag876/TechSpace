public class WeatherActivitySuggester
{
    public class Weather
    {
        public int Temperature { get; set; }
        public string Condition { get; set; } // Sunny, Rainy, Snowy, Cloudy
        public int Humidity { get; set; }
        public int WindSpeed { get; set; }
    }
    
    public string SuggestActivity(Weather weather)
    {
        if (weather.Temperature >= 25 && weather.Condition == "Sunny")
        {
            if (weather.Humidity < 40)
            {
                return "Perfect for hiking or rock climbing!";
            }
            else
            {
                return "Great for swimming or beach day!";
            }
        }
        else if (weather.Temperature >= 15 && weather.Temperature < 25)
        {
            if (weather.Condition == "Sunny")
            {
                return "Ideal for cycling or outdoor photography!";
            }
            else if (weather.Condition == "Cloudy")
            {
                return "Good for running or jogging!";
            }
        }
        else if (weather.Temperature < 15)
        {
            if (weather.Condition == "Snowy")
            {
                return "Time for skiing or snowboarding!";
            }
            else if (weather.Condition == "Rainy" && weather.WindSpeed > 30)

                return "Stay home, watch movies, drink hot chocolate!";

            else if (weather.Condition == "Rainy")
            {

The Switch Statement

Used when you have multiple specific values to check against.

Basic Switch

public class SwitchExample
{
    // Day of week checker
    public void DayOfWeekChecker(int dayNumber)
    {
        switch (dayNumber)
        {
            case 1:
                Console.WriteLine("Monday - Start of work week");
                break;
            case 2:
                Console.WriteLine("Tuesday - Productivity peak");
                break;
            case 3:
                Console.WriteLine("Wednesday - Hump day");
                break;
            case 4:
                Console.WriteLine("Thursday - Almost there");
                break;
            case 5:
                Console.WriteLine("Friday - TGIF!");
                break;
            case 6:
            case 7:
                Console.WriteLine("Weekend time!");
                break;
            default:
                Console.WriteLine("Invalid day number");
                break;
        }
    }
    
    // Without break - Fall through (rare, intentional)
    public void PermissionChecker(string role)
    {
        bool canRead = false;
        bool canWrite = false;
        bool canDelete = false;
        
        switch (role)
        {
            case "Admin":
                canDelete = true;
                goto case "Manager"; // Fall through to Manager
            case "Manager":
                canWrite = true;
                goto case "User"; // Fall through to User
            case "User":
                canRead = true;
                break;
            default:
                Console.WriteLine("Unknown role");
                break;
        }
        
        Console.WriteLine($"Read: {canRead}, Write: {canWrite}, Delete: {canDelete}");
    }
}

public class GameDifficultySelector
{
    public class GameLevel
    {
        public string PlayerSkill { get; set; } // Beginner, Intermediate, Advanced, Expert

public class TernaryExample
{
    public void BasicTernary()
    {

        string status = (age >= 18) ? "Adult" : "Minor";
        Console.WriteLine(status); // Adult
    }
    
    // Nested ternary (not recommended for readability)
    public string GradeWithTernary(int marks)
    {
        return marks >= 90 ? "A" : 
               marks >= 80 ? "B" : 
               marks >= 70 ? "C" : 
               marks >= 60 ? "D" : "F";
    }
    
    // Real-world: Pricing calculator
    public decimal CalculatePrice(int quantity, bool isPremiumMember)
    {
        decimal basePrice = 100m;
        
        // Quantity discount
        decimal unitPrice = quantity > 100 ? basePrice * 0.8m :
                           quantity > 50 ? basePrice * 0.9m :
                           basePrice;
        
        // Member discount
        unitPrice = isPremiumMember ? unitPrice * 0.95m : unitPrice;
        
        return unitPrice * quantity;
    }

Logical Operators with Conditional Statements

Combining multiple conditions.

public class LogicalOperatorsExample
{
    public class User
    {
        public int Age { get; set; }
        public string Country { get; set; }
        public bool EmailVerified { get; set; }
        public bool PhoneVerified { get; set; }
        public int AccountAgeInDays { get; set; }
    }
    
    // AND operator (&&) - All must be true
    public bool CanAccessAdultContent(User user)
    {
        if (user.Age >= 18 && user.EmailVerified && user.PhoneVerified)
        {
            return true;
        }
        return false;
    }
    
    // OR operator (||) - At least one must be true
    public bool QualifiesForDiscount(User user)
    {
        if (user.AccountAgeInDays > 365 || user.Country == "India")
        {
            return true;
        }
        return false;
    }
    
    // NOT operator (!) - Inverts condition
    public bool CanModifyContent(User user)
    {
        if (!user.EmailVerified)
        {
            return false; // Must verify email first
        }
        return true;
    }
    
    // Complex logic: Bank account eligibility
    public bool EligibleForLoan(User user)
    {
        bool isOfAge = user.Age >= 21;
        bool isVerified = user.EmailVerified && user.PhoneVerified;
        bool hasGoodAccountAge = user.AccountAgeInDays > 180;
        bool isInEligibleCountry = user.Country == "India" || user.Country == "USA";
        
        if (isOfAge && isVerified && (hasGoodAccountAge || isInEligibleCountry))
        {
            return true;
        }
        return false;
    }
    
    // Short-circuit evaluation example
    public void ShortCircuitDemo()
    {
        int x = 0;
        
        // If first condition is false, second is not evaluated (short-circuit)
        if (x > 0 && DangerousOperation(x))
        {
            Console.WriteLine("Safe");
        }
        // DangerousOperation won't be called because x > 0 is false
        
        if (x > 0 || SafeOperation(x))
        {
            Console.WriteLine("Executed");
        }
        // SafeOperation will be called because x > 0 is false
    }
    
    private bool DangerousOperation(int value)
    {
        return value > 5;
    }
    
    private bool SafeOperation(int value)
    {
        return true;
    }
}

Loop Types

The For Loop

Most controlled loop, perfect when you know iteration count.

Basic For Loop

public class ForLoopExample
{
    // Simple iteration
    public void PrintNumbers()
    {
        for (int i = 0; i < 10; i++)
        {
            Console.WriteLine(i); // 0 to 9
        }
    }
    
    // Reverse iteration
    public void PrintReverse()
    {
        for (int i = 10; i > 0; i--)
        {
            Console.WriteLine(i); // 10 to 1
        }
    }
    
    // Multiple variables
    public void MultipleVariables()
    {
        for (int i = 0, j = 10; i < 10; i++, j--)
        {
            Console.WriteLine($"i={i}, j={j}"); // i increases, j decreases
        }
    }
    
    // Nested for loop - multiplication table
    public void MultiplicationTable()
    {
        for (int i = 1; i <= 10; i++)
        {
            for (int j = 1; j <= 10; j++)
            {
                Console.Write($"{i * j,4} "); // 4 characters wide
            }
            Console.WriteLine();
        }
    }
}

Interesting Real-World Example: Password Strength Checker

public class PasswordStrengthChecker
{
    public enum PasswordStrength
    {
        VeryWeak = 1,
        Weak = 2,
        Fair = 3,
        Good = 4,
        Strong = 5,
        VeryStrong = 6
    }
    
    public class PasswordAnalysis
    {
        public PasswordStrength Strength { get; set; }
        public int Score { get; set; }
        public List<string> Criteria { get; set; }
        public List<string> Suggestions { get; set; }
    }
    
    public PasswordAnalysis AnalyzePassword(string password)
    {
        var analysis = new PasswordAnalysis
        {
            Criteria = new List<string>(),
            Suggestions = new List<string>(),
            Score = 0
        };
        
        // Check each criterion using for loop to understand the password character by character
        bool hasLowercase = false;
        bool hasUppercase = false;
        bool hasDigits = false;
        bool hasSpecialChars = false;
        
        // Iterate through each character
        for (int i = 0; i < password.Length; i++)
        {
            char c = password[i];
            
            if (char.IsLower(c))
                hasLowercase = true;
            if (char.IsUpper(c))
                hasUppercase = true;
            if (char.IsDigit(c))
                hasDigits = true;
            if (!char.IsLetterOrDigit(c))
                hasSpecialChars = true;
        }
        
        // Length check
        if (password.Length >= 8)
        {
            analysis.Score += 10;
            analysis.Criteria.Add("Length >= 8 characters");
        }
        else
        {
            analysis.Suggestions.Add("Use at least 8 characters");
        }
        
        if (password.Length >= 12)
        {
            analysis.Score += 10;
            analysis.Criteria.Add("Length >= 12 characters");
        }
        
        if (password.Length >= 16)
        {
            analysis.Score += 10;
            analysis.Criteria.Add("Length >= 16 characters");
        }
        
        // Check lowercase
        if (hasLowercase)
        {
            analysis.Score += 15;
            analysis.Criteria.Add("Contains lowercase letters");
        }
        else
        {
            analysis.Suggestions.Add("Add lowercase letters (a-z)");
        }
        
        // Check uppercase
        if (hasUppercase)
        {
            analysis.Score += 15;
            analysis.Criteria.Add("Contains uppercase letters");
        }
        else
        {
            analysis.Suggestions.Add("Add uppercase letters (A-Z)");
        }
        
        // Check digits
        if (hasDigits)
        {
            analysis.Score += 20;
            analysis.Criteria.Add("Contains numbers");
        }
        else
        {
            analysis.Suggestions.Add("Add numbers (0-9)");
        }
        
        // Check special characters
        if (hasSpecialChars)
        {
            analysis.Score += 25;
            analysis.Criteria.Add("Contains special characters");
        }
        else
        {
            analysis.Suggestions.Add("Add special characters (!@#$%^&*)");
        }
        
        // Determine strength
        if (analysis.Score >= 90)
            analysis.Strength = PasswordStrength.VeryStrong;
        else if (analysis.Score >= 75)
            analysis.Strength = PasswordStrength.Strong;
        else if (analysis.Score >= 60)
            analysis.Strength = PasswordStrength.Good;
        else if (analysis.Score >= 45)
            analysis.Strength = PasswordStrength.Fair;
        else if (analysis.Score >= 20)
            analysis.Strength = PasswordStrength.Weak;
        else
            analysis.Strength = PasswordStrength.VeryWeak;
        
        return analysis;
    }
    
    // Usage
    public void Demo()
    {
        string password = "MySecure@Pass123";
        var result = AnalyzePassword(password);
        
        Console.WriteLine($"Password Strength: {result.Strength}");
        Console.WriteLine($"Score: {result.Score}/100");
        Console.WriteLine("\nCriteria Met:");
        foreach (var criterion in result.Criteria)
        {
            Console.WriteLine($"  ✓ {criterion}");
        }
        
        if (result.Suggestions.Count > 0)
        {
            Console.WriteLine("\nSuggestions:");
            foreach (var suggestion in result.Suggestions)
            {
                Console.WriteLine($"  - {suggestion}");
            }
        }
    }
}

Interesting Real-World Example: Login Retry System

public class LoginRetrySystem
{
    public class LoginAttempt
    {
        public string Username { get; set; }
        public string Password { get; set; }
        public int AttemptNumber { get; set; }
        public bool Success { get; set; }
        public string Message { get; set; }
    }
    
    private Dictionary<string, string> validCredentials = new()
    {
        { "user1", "pass123" },
        { "user2", "pass456" },
        { "admin", "admin@123" }
    };
    
    private Dictionary<string, int> failedAttempts = new();
    private const int MAX_ATTEMPTS = 3;
    private const int LOCKOUT_MINUTES = 15;
    
    public LoginAttempt AttemptLogin(string username)
    {
        var result = new LoginAttempt { Username = username, AttemptNumber = 0 };
        
        // Check if account is locked
        if (failedAttempts.ContainsKey(username) && failedAttempts[username] >= MAX_ATTEMPTS)
        {
            result.Message = $"Account locked. Try again after {LOCKOUT_MINUTES} minutes.";
            return result;
        }
        
        // Allow user to retry
        int attemptCount = 0;
        while (attemptCount < MAX_ATTEMPTS)
        {
            attemptCount++;
            result.AttemptNumber = attemptCount;
            
            Console.Write($"Attempt {attemptCount}/{MAX_ATTEMPTS}\n");
            Console.Write("Enter password: ");
            string password = Console.ReadLine();
            
            if (validCredentials.ContainsKey(username) && validCredentials[username] == password)
            {
                result.Success = true;
                result.Message = $"Login successful! Welcome {username}";
                failedAttempts[username] = 0; // Reset failed attempts
                return result;
            }
            else
            {
                int remainingAttempts = MAX_ATTEMPTS - attemptCount;
                if (remainingAttempts > 0)
                {
                    result.Message = $"Invalid password. {remainingAttempts} attempts remaining.";
                    Console.WriteLine(result.Message);
                }
                else
                {
                    failedAttempts[username] = MAX_ATTEMPTS;
                    result.Message = "Maximum attempts exceeded. Account locked.";
                }
            }
        }
        
        return result;
    }
    
    // Usage
    public void Demo()
    {
        var loginResult = AttemptLogin("user1");
        Console.WriteLine(loginResult.Message);
    }
}

Interesting Real-World Example: ATM Machine Simulator

public class ATMMachineSimulator
{
    public class Account
    {
        public string AccountNumber { get; set; }
        public decimal Balance { get; set; }
        public string PIN { get; set; }
        public List<string> Transactions { get; set; }
    }
    
    private Dictionary<string, Account> accounts = new()
    {
        { "1001", new Account { AccountNumber = "1001", Balance = 50000, PIN = "1234", Transactions = new List<string>() } },
        { "1002", new Account { AccountNumber = "1002", Balance = 100000, PIN = "5678", Transactions = new List<string>() } }
    };
    
    public void RunATM(string accountNumber)
    {
        if (!accounts.ContainsKey(accountNumber))
        {
            Console.WriteLine("Account not found");
            return;
        }
        
        Account account = accounts[accountNumber];
        
        // Verify PIN
        Console.Write("Enter PIN: ");
        string enteredPIN = Console.ReadLine();
        
        if (enteredPIN != account.PIN)
        {
            Console.WriteLine("Incorrect PIN");
            return;
        }
        
        int choice;
        do
        {
            Console.WriteLine("\n========== ATM MENU ==========");
            Console.WriteLine($"Current Balance: Rs {account.Balance}");
            Console.WriteLine("1. Withdraw");
            Console.WriteLine("2. Deposit");
            Console.WriteLine("3. Check Balance");
            Console.WriteLine("4. Transaction History");
            Console.WriteLine("5. Exit");
            Console.Write("Select Option: ");
            
            int.TryParse(Console.ReadLine(), out choice);
            
            switch (choice)
            {
                case 1: // Withdraw
                    Console.Write("Enter amount to withdraw: Rs ");
                    if (decimal.TryParse(Console.ReadLine(), out decimal withdrawAmount))
                    {
                        if (withdrawAmount > account.Balance)
                        {
                            Console.WriteLine("Insufficient balance!");
                        }
                        else if (withdrawAmount <= 0)
                        {
                            Console.WriteLine("Invalid amount!");
                        }
                        else
                        {
                            account.Balance -= withdrawAmount;
                            account.Transactions.Add($"Withdrawal: Rs {withdrawAmount} - Balance: Rs {account.Balance}");
                            Console.WriteLine($"Amount withdrawn successfully. New balance: Rs {account.Balance}");
                        }
                    }
                    break;
                
                case 2: // Deposit
                    Console.Write("Enter amount to deposit: Rs ");
                    if (decimal.TryParse(Console.ReadLine(), out decimal depositAmount))
                    {
                        if (depositAmount <= 0)
                        {
                            Console.WriteLine("Invalid amount!");
                        }
                        else
                        {
                            account.Balance += depositAmount;
                            account.Transactions.Add($"Deposit: Rs {depositAmount} - Balance: Rs {account.Balance}");
                            Console.WriteLine($"Amount deposited successfully. New balance: Rs {account.Balance}");
                        }
                    }
                    break;
                
                case 3: // Check Balance
                    Console.WriteLine($"\nYour current balance: Rs {account.Balance}");
                    break;
                
                case 4: // Transaction History
                    Console.WriteLine("\n=== Transaction History ===");
                    if (account.Transactions.Count == 0)
                    {
                        Console.WriteLine("No transactions yet.");
                    }
                    else
                    {
                        for (int i = 0; i < account.Transactions.Count; i++)
                        {
                            Console.WriteLine($"{i + 1}. {account.Transactions[i]}");
                        }
                    }
                    break;
                
                case 5: // Exit
                    Console.WriteLine("\nThank you for using our ATM. Please take your card.");
                    break;
                
                default:
                    Console.WriteLine("Invalid option. Try again.");
                    break;
            }
        } while (choice != 5);
    }
    
    // Usage
    public void Demo()
    {
        RunATM("1001");
    }
}

public class BreakExample

    public void SearchArray()
    {
        int[] numbers = { 10, 20, 30, 40, 50 };
        int searchValue = 30;
        
        for (int i = 0; i < numbers.Length; i++)
        {
            if (numbers[i] == searchValue)

                break; // Exit loop when found
            }
        }
    }
    
    // Break in while loop
    public void FindUserInList()
    {
        List<string> users = new() { "Alice", "Bob", "Charlie", "Diana" };
        string searchUser = "Charlie";
        int index = 0;
        
        while (index < users.Count)
        {
            if (users[index] == searchUser)
            {
                Console.WriteLine($"{searchUser} found at index {index}");
                break;
            }
            index++;
        }
    }
    
    // Break in switch
    public void DayOfWeek(int day)
    {
        switch (day)
        {
            case 1:
                Console.WriteLine("Monday");
                break; // Must break to avoid fall-through
            case 2:
                Console.WriteLine("Tuesday");
                break;
        }
    }

    {
        private List<Book> books = new();

        {
            books.Add(book);
        }
        
        // Search by title with early exit
        public Book SearchByTitle(string searchTitle)
        {
            Console.WriteLine($"Searching for book: {searchTitle}...");
            
            for (int i = 0; i < books.Count; i++)
            {
                Console.WriteLine($"  Checking index {i}: {books[i].Title}");
                
                if (books[i].Title.ToLower().Contains(searchTitle.ToLower()))
                {
                    Console.WriteLine($"Found at index {i}!");
                    return books[i]; // Break implicitly via return
                }
            }
            
            Console.WriteLine("Book not found!");
            return null;
        }
        
        // Search with explicit break
        public Book SearchAndReserve(string searchTitle)
        {
            Book foundBook = null;
            
            for (int i = 0; i < books.Count; i++)
            {
                if (books[i].Title.Equals(searchTitle, StringComparison.OrdinalIgnoreCase))
                {
                    if (books[i].IsAvailable)
                    {
                        foundBook = books[i];
                        break; // Exit loop once found
                    }
                }
            }
            
            if (foundBook != null)
            {
                foundBook.IsAvailable = false;
                Console.WriteLine($"Book '{foundBook.Title}' reserved successfully!");
            }
            else
            {
                Console.WriteLine("Book not available for reservation.");
            }
            
            return foundBook;
        }
    }

    public void Demo()
    {

public class DataCleaningPipeline
{
    public class SensorReading
    {
        public int ReadingId { get; set; }
        public decimal TemperatureInCelsius { get; set; }
        public decimal Humidity { get; set; }
        public string Status { get; set; }
    }
    
    public class DataValidator
    {
        public List<SensorReading> readings = new();
        
        public void AddReading(SensorReading reading)
        {
            readings.Add(reading);
        }
        
        // Process only valid readings
        public void GenerateWeatherReport()
        {
            Console.WriteLine("\n========== WEATHER REPORT ==========");
            decimal totalTemp = 0;
            int validReadings = 0;
            
            foreach (SensorReading reading in readings)
            {
                // Skip invalid readings
                if (reading.TemperatureInCelsius < -50 || reading.TemperatureInCelsius > 60)
                {
                    Console.WriteLine($"[SKIPPED] Reading {reading.ReadingId}: Temperature out of range ({reading.TemperatureInCelsius}°C)");
                    continue; // Skip this reading
                }
                
                if (reading.Humidity < 0 || reading.Humidity > 100)
                {
                    Console.WriteLine($"[SKIPPED] Reading {reading.ReadingId}: Invalid humidity ({reading.Humidity}%)");
                    continue; // Skip this reading
                }
                
                // Process valid readings
                totalTemp += reading.TemperatureInCelsius;
                validReadings++;
                
                Console.WriteLine($"[VALID] Reading {reading.ReadingId}: {reading.TemperatureInCelsius}°C, {reading.Humidity}% humidity");
            }
            
            if (validReadings > 0)
            {
                decimal averageTemp = totalTemp / validReadings;
                Console.WriteLine($"\nAverage Temperature (valid readings): {averageTemp:F2}°C");
            }
        }
    }
    
    // Usage
    public void Demo()
    {
        var validator = new DataValidator();
        
        validator.AddReading(new SensorReading { ReadingId = 1, TemperatureInCelsius = 25.5m, Humidity = 65, Status = "OK" });
        validator.AddReading(new SensorReading { ReadingId = 2, TemperatureInCelsius = -100, Humidity = 50, Status = "ERROR" }); // Out of range
        validator.AddReading(new SensorReading { ReadingId = 3, TemperatureInCelsius = 30.2m, Humidity = 75, Status = "OK" });
        validator.AddReading(new SensorReading { ReadingId = 4, TemperatureInCelsius = 22, Humidity = 150, Status = "ERROR" }); // Invalid humidity
        validator.AddReading(new SensorReading { ReadingId = 5, TemperatureInCelsius = 28.7m, Humidity = 60, Status = "OK" });

public class StudentGradeManagementSystem

    {

        {
            if (students[i].StudentId == studentId)
            {
                if (marks < 0 || marks > 100)
                {

            }

            return null;

        decimal sum = 0;
        foreach (int mark in student.Marks)
        {
            sum += mark;

Interview Questions

Conditional Statement Questions

Q1: What’s the difference between if-else and switch statements?
A: if-else evaluates conditions sequentially, works with ranges. switch checks exact values against cases, more efficient for multiple specific values. switch jumps directly to matching case using jump table.

Q2: Why use ternary operator? When should you avoid it?
A: Ternary is concise for simple two-way decisions. Avoid nested ternary (unreadable). Max one level deep. Use if-else for complex logic.

Q3: What does short-circuit evaluation mean?
A: In (A && B), if A is false, B isn’t evaluated. In (A || B), if A is true, B isn’t evaluated. Improves performance and prevents null reference exceptions.

Q4: Can you use break inside nested if statements?
A: No, break only works inside loops and switch. For nested if, use return, goto, or restructure logic.

Loop Questions

Q1: When would you choose for, while, and foreach loops?
A: for when iteration count is known. while when count is unknown or waiting for condition. foreach for iterating collections (cleaner, no index errors).

Q2: What’s the difference between break and continue?
A: break exits loop entirely. continue skips current iteration, continues with next. break = exit, continue = skip.

Q3: Can foreach loop modify collection while iterating?
A: No, you’ll get InvalidOperationException. Use for loop or create new list if modification needed.

Q4: Explain do-while loop usage.
A: Executes body at least once before checking condition. Used for menus, input validation where you need minimum one execution.

Control Flow Questions

Q1: What’s the downside of goto statement?
A: Makes code hard to follow (spaghetti code). Jumps around unpredictably. Modern languages avoid it. Use it sparingly for error handling only.

Q2: Can you have multiple breaks in one loop?
A: Yes, break at multiple conditions. Each break exits at that point. Each needs its own if condition.

Q3: What happens with nested loops and break?
A: break only exits innermost loop. To exit outer loop, use flag variable, goto, or return.

Quick Reference Cheat Sheet

If-Else Pattern

if (condition)
    // execute if true
else if (anotherCondition)
    // execute if second condition true
else
    // execute if all false

Switch Pattern

switch (variable)
{
    case value1:
        // execute
        break;
    case value2:
        // execute
        break;
    default:
        // execute if no match
        break;
}

For Loop Pattern

for (initialization; condition; increment)
{
    // execute while condition true
}

While Loop Pattern

while (condition)
{
    // execute while condition true
}

Do-While Loop Pattern

do
{
    // execute at least once
} while (condition);

Foreach Loop Pattern

foreach (type item in collection)
{
    // execute for each item
}

Break and Continue

break;      // Exit loop immediately
continue;   // Skip to next iteration
goto Label; // Jump to labeled location

Summary

Master these control structures:

  1. Conditional Statements: if-else for ranges, switch for exact values, ternary for simple decisions
  2. Loops: for (known count), while (unknown count), do-while (minimum once), foreach (collections)
  3. Control Flow: break (exit), continue (skip), goto (jump - use rarely)
  4. Real-world patterns: Validation, search, filtering, processing
  5. Performance: Use appropriate loop type, early exit with break, skip with continue

Practice combining these patterns in complex scenarios. Good code uses clear control flow without excessive nesting or goto statements.
